class Solution {
    public int countStudents(int[] students, int[] sandwiches) {
        int[] counts = new int[2]; // counts[0] for type 0, counts[1] for type 1
        
        // Step 1: Count preferences
        for (int student : students) {
            counts[student]++;
        }
        
        // Step 2: Try to satisfy each sandwich in order
        for (int sandwich : sandwiches) {
            if (counts[sandwich] > 0) {
                counts[sandwich]--;
            } else {
                // No students left who want this type of sandwich
                // The stack is stuck, and everyone remaining will go hungry
                break;
            }
        }
        
        // Step 3: Remaining students are the ones who couldn't eat
        return counts[0] + counts[1];
    }
}
