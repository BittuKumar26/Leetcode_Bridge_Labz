class Solution {
    public ListNode removeNthFromEnd(ListNode head, int n) {
        if(head == null) return null;
        ListNode temp = head;
        int cnt = 0;
        while(temp!=null){
            cnt++;
            temp = temp.next;
        }
        cnt = cnt - n + 1;
        temp = head;
        if(cnt==1) return head.next;
        for(int i=1;i<cnt-1;i++){
            temp = temp.next;
        }
        if(temp==null || temp.next==null) return null;
        temp.next = temp.next.next;
        return head;
    }
}
